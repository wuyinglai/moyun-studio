| 类型 | 规范 | 示例 |
|------|------|------|
| 文件名 | snake_case | `file_ops.py`, `llm_config.py` |
| 类名 | PascalCase | `LLMService`, `FileWatcher` |
| 函数/方法 | snake_case | `read_file()`, `count_tokens()` |
| 常量 | UPPER_SNAKE_CASE | `MAX_RETRIES`, `DEFAULT_TIMEOUT` |
| 私有属性 | _前缀 | `_internal_state` |
| 实例变量 | snake_case | `self.task_queue` |
| 模块名 | snake_case | `utils`, `core` |

## 三、异步规范

### 3.1 基本原则

```python
# ✅ 正确：使用 async/await
async def read_file(self, path: str) -> str:
    async with aiofiles.open(path, 'r', encoding='utf-8') as f:
        return await f.read()

# ❌ 错误：禁止同步操作
def read_file(self, path: str) -> str:
    with open(path, 'r') as f:  # 禁止！
        return f.read()
```

### 3.2 异步函数定义

```python
# ✅ 必须使用 async def
async def fetch_data(self, url: str) -> dict:
    ...

# ❌ 不要混用
def fetch_data(self, url: str) -> dict:  # 禁止！
    ...
```

### 3.3 异步生成器

```python
# ✅ 流式输出使用异步生成器
async def stream_generate(
    self,
    messages: list[dict]
) -> AsyncGenerator[str, None]:
    async for chunk in self.client.stream(messages):
        yield chunk
```

## 五、异常处理

### 5.1 自定义异常类

```python
# ✅ 定义清晰的自定义异常
class AppException(Exception):
    """应用异常基类"""
    def __init__(self, message: str, code: str):
        self.message = message
        self.code = code
        super().__init__(self.message)

class FileNotFoundError(AppException):
    """文件不存在"""
    def __init__(self, path: str):
        super().__init__(
            message=f"文件不存在: {path}",
            code="FILE_NOT_FOUND"
        )

class TemplateNotFoundError(AppException):
    """模板不存在"""
    def __init__(self, template: str):
        super().__init__(
            message=f"模板不存在: {template}",
            code="TEMPLATE_NOT_FOUND"
        )

class LLMError(AppException):
    """LLM调用错误"""
    def __init__(self, message: str, model: str | None = None):
        detail = f"{message} (model: {model})" if model else message
        super().__init__(
            message=f"LLM调用失败: {detail}",
            code="LLM_ERROR"
        )
```

### 5.2 异常捕获与重新抛出

```python
# ✅ 使用 try-except 并重新抛出有意义异常
async def read_file(self, path: str) -> str:
    try:
        async with aiofiles.open(path, 'r', encoding='utf-8') as f:
            return await f.read()
    except FileNotFoundError:
        raise FileNotFoundError(path)
    except PermissionError:
        raise AppException(
            message=f"权限不足: {path}",
            code="PERMISSION_DENIED"
        )
    except Exception as e:
        raise AppException(
            message=f"读取文件失败: {e}",
            code="FILE_READ_ERROR"
        )
```

### 5.3 异常日志

```python
# ✅ 记录异常堆栈
try:
    await self.process(data)
except Exception as e:
    logger.error(
        f"处理数据失败: {e}",
        exc_info=True,  # 包含完整堆栈
        extra={"task_id": self.task_id}
    )
    raise
```

## 七、配置管理

### 7.1 pydantic-settings 使用

```python
# ✅ 使用 pydantic-settings
from pydantic_settings import BaseSettings, SettingsConfigDict

class Settings(BaseSettings):
    """应用配置"""
    model_config = SettingsConfigDict(
        env_file=".env",
        env_file_encoding="utf-8",
        env_prefix="MOYUN_",
        case_sensitive=False
    )

    # 必填项：无默认值
    llm_api_key: str

    # 可选项：有默认值
    llm_model: str = "gpt-4"
    workspace_path: str = "./workspace"
    max_retries: int = 3

    # 敏感字段
    @property
    def masked_api_key(self) -> str:
        """返回脱敏的 API Key"""
        if len(self.llm_api_key) > 8:
            return f"{self.llm_api_key[:4]}...{self.llm_api_key[-4:]}"
        return "***"

# ❌ 禁止使用 os.getenv
api_key = os.getenv("API_KEY")  # 禁止！
```

## 八、API路由规范

### 8.1 路由文件结构

```python
# api/projects.py
from fastapi import APIRouter, HTTPException, Depends
from pydantic import BaseModel, Field
from typing import Optional

router = APIRouter(prefix="/projects", tags=["项目管理"])

# ============ 请求/响应模型 ============

class CreateProjectRequest(BaseModel):
    """创建项目请求"""
    name: str = Field(..., min_length=1, max_length=100)
    genre: str
    theme: str
    scale: int = Field(default=50000, ge=10000, le=200000)

class ProjectResponse(BaseModel):
    """项目响应"""
    project_id: str
    name: str
    genre: str

# ============ 依赖注入 ============

async def get_project_service() -> ProjectService:
    """获取项目服务"""
    return ProjectService(file_service=app.state.file_service)

# ============ 路由定义 ============

@router.post("/", response_model=ProjectResponse, status_code=201)
async def create_project(
    req: CreateProjectRequest,
    service: ProjectService = Depends(get_project_service)
):
    """创建新项目

    创建一个新的小说项目，包含基础目录结构。

    Args:
        req: 创建项目请求参数
        service: 项目服务

    Returns:
        创建的项目信息

    Raises:
        HTTPException: 项目创建失败时抛出
    """
    try:
        project = await service.create(req)
        return ProjectResponse(
            project_id=project.id,
            name=project.name,
            genre=project.genre
        )
    except ValueError as e:
        raise HTTPException(status_code=400, detail=str(e))
```

### 8.2 响应格式

```python
# ✅ 使用统一响应格式
from typing import Generic, TypeVar

T = TypeVar("T")

class ApiResponse(BaseModel, Generic[T]):
    """统一API响应"""
    success: bool
    data: T | None = None
    message: str | None = None
    error: dict | None = None

def success_response(data: Any = None, message: str = "操作成功") -> dict:
    return ApiResponse(success=True, data=data, message=message).model_dump()

def error_response(message: str, code: str, details: Any = None) -> dict:
    return ApiResponse(
        success=False,
        error={"code": code, "details": details}
    ).model_dump()
```

## 十、依赖注入

### 10.1 服务依赖

```python
# ✅ 使用 FastAPI 依赖注入
from fastapi import Depends

async def get_llm_service() -> LLMService:
    return app.state.llm

async def get_file_service() -> FileService:
    return app.state.file_service

async def get_prompt_engine(
    file_service: FileService = Depends(get_file_service)
) -> PromptEngine:
    return PromptEngine(
        prompts_path=settings.prompts_path,
        file_service=file_service
    )

# ✅ 在路由中使用
@router.post("/generate")
async def generate_content(
    req: GenerateRequest,
    llm: LLMService = Depends(get_llm_service),
    engine: PromptEngine = Depends(get_prompt_engine)
):
    prompt = await engine.render(req.template, req.variables)
    return await llm.complete(prompt)
```

### 10.2 避免循环依赖

```python
# ❌ 避免循环依赖
# a.py
from .b import B
class A:
    def __init__(self):
        self.b = B()

# b.py
from .a import A  # 可能导致循环导入！
class B:
    def __init__(self):
        self.a = A()

# ✅ 使用延迟导入或依赖注入
class B:
    def __init__(self, a: "A" = None):  # 类型注解用字符串
        self.a = a
```

### 13.2 函数文档字符串

```python
async def complete(
    self,
    messages: list[dict[str, str]],
    model: str | None = None,
    stream: bool = True,
    temperature: float = 0.7
) -> AsyncGenerator[str, None]:
    """流式生成文本

    根据对话历史生成回复，支持流式和非流式两种模式。

    Args:
        messages: 对话消息列表，每条消息包含 role 和 content
        model: 模型名称，None使用默认模型
        stream: 是否流式输出，默认True
        temperature: 温度参数，控制随机性

    Yields:
        生成的文本片段（流式模式）

    Returns:
        完整生成的文本（非流式模式）

    Raises:
        LLMError: LLM调用失败时抛出
        ValidationError: 参数验证失败时抛出

    Example:
        >>> messages = [{"role": "user", "content": "你好"}]
        >>> async for chunk in service.complete(messages):
        ...     print(chunk, end="")
        你好！有什么可以帮助你的吗？
    """
```
